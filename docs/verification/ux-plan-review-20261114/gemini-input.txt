## 5. Frontend và backend: đã nối (14/11)

- Định dạng dữ liệu của frontend nằm ở `water-pulse-fe/src/data/types.ts`. Backend phục vụ đúng định dạng đó qua `water-pulse-be/app/frontend.py` (`/api/frontend/scenes|communes|communes/{id}/series|choropleth|area|summary|validation`).
- **Trạng thái xã:** `ready`→`receded`, `draining`→`uncertain`, `still_wet`→`not_receded`. Thêm `no_flood` khi đỉnh < 5%. Xã không đủ ảnh phủ thì `uncertain` kèm `coverage_sufficient=false`.
- **Số liệu thống kê dùng một track tham chiếu**, là track của ảnh gần nhất phủ ít nhất nửa xã (`reference_track`). Chuỗi thời gian giữ riêng từng track.
- **Cố ý để trống:**
  - `dswx_km2` là `null`, vì chưa có tổng DSWx trên cùng vùng;
  - `summary` là `[]`, vì ảnh tổng kết hiện gộp các track. Đạt đang xuất bản theo từng track.
- **Khi `data/web` thay đổi:** chạy `npm run build:cloudflare`, `npm run test:cloudflare`, rồi deploy lại.

## 8. Bẫy kỹ thuật đã gặp (đừng lặp lại)

- **NASA GIBS, lớp `NISAR_L2_Geocoded_Polarimetric_Covariance`:**
  - zoom **11–13 trả về tile hỏng**, nên đặt `maxzoom: 10`;
  - ngoài dải quét GIBS trả HTTP 500;
  - có thể thay bằng `data/web/radar/<date>_<track>.jpg` (khoảng 60 m, cùng `bounds` với ảnh lớp nước, khớp từng pixel).
- **MapLibre GL 6:**
  - cần `setWorkerUrl` và `optimizeDeps.exclude` trong Vite, thiếu thì GeoJSON không tải;
  - MapLibre dùng tile 512 nên `zoom` lệch 1 mức so với Leaflet.
- **Nền bản đồ:** CARTO miễn phí giờ đòi API key, frontend đang dùng Esri World Light Gray.
- **GCOV:**
  - giá trị là γ⁰ dạng công suất tuyến tính (đổi sang dB = 10·log10);
  - `mask` 0 hoặc 255 là pixel không hợp lệ;
  - lấy trung bình trong miền công suất, không lấy trung bình trên dB;
  - URL tải là link CloudFront có chữ ký, mỗi link chỉ resolve một lần.
- **Ngưỡng tách nước:**
  - Otsu theo từng ngày cho kết quả sai, nên dùng ngưỡng cố định đã hiệu chỉnh;
  - track 148D không có DSWx-S1 cùng ngày để so;
  - chỉ so với DSWx-HLS trong mùa lũ (từ 01/08), vì ảnh tháng 6–7 nhiều mây và xử lý chậm.
- **Overpass API:** máy chủ chính hay timeout, dùng mirror `overpass.kumi.systems` và Nominatim.
- **Kiểm tra xuống giống:**
  - lấy **ảnh gần nhất phủ ≥ 50% xã** (không lấy trường hợp xấu nhất của mọi track, vì cách đó gây sai, ví dụ ở Tân Hồng);
  - ngưỡng: < 10% là sẵn sàng, 10–30% là đang rút, ≥ 30% là còn ngập.
- **Web research:**
  - API của editor (BlockNote) từ chối text vừa **đậm vừa code**;
  - trạng thái task chỉ nhận `todo | blocked | done`;
  - đăng bài qua `POST /api/ingest` (cần key; hướng dẫn ở `skyradar-research/docs/AI_INGEST_PROMPT.md`).

===== DỮ LIỆU NGUỒN: /Users/Chris/nisar/skyradar-nisar/src/skyradar_nisar/sowing.py =====
"""Sowing-readiness check per commune: water on the latest image of each track around the sowing window.

    python -m skyradar_nisar.sowing [YYYY-MM-DD]     # reference date, default 2026-11-01 (An Giang winter-spring window)
Outputs: data/processed/stats/sowing_check.csv, docs/figures/fig5_sowing_check.png
Status rule (per commune): the MOST RECENT image among tracks covering >= 50 % of the commune (worst case across tracks kept as worst_*):
  still_wet  >= 30 % of the commune under water on the latest image
  draining   10-30 %
  ready      < 10 %
"""
import csv, json, sys
import numpy as np
import matplotlib; matplotlib.use("Agg")
import matplotlib.pyplot as plt
from matplotlib.patches import Patch
from shapely.geometry import shape
from . import config as C

LEVELS = [("ready", 0, 10, "#2f9e44"), ("draining", 10, 30, "#f59f00"), ("still_wet", 30, 101, "#7048e8")]


def status(p):
    return next(name for name, lo, hi, _ in LEVELS if lo <= p < hi)


def main(ref="2026-11-01"):
    rows = list(csv.DictReader(open(C.PROCESSED / "stats/commune_water_by_date.csv", encoding="utf-8")))
    latest = {}
    for r in rows:
        if float(r["valid_km2"]) < 5:
            continue
        k = (r["osm_id"], r["track"])
        if k not in latest or r["date"] > latest[k]["date"]:
            latest[k] = r
    out = {}
    for (oid, trk), r in latest.items():
        e = out.setdefault(oid, dict(osm_id=oid, commune=r["commune"], province=r["province"], per_track={}))
        e["per_track"][trk] = (r["date"], float(r["water_pct"]), float(r["valid_km2"]))
    table = []
    for e in out.values():
        pt = e["per_track"]
        worst_trk, (wd, wp, _) = max(pt.items(), key=lambda kv: kv[1][1])
        # status = most recent image among tracks that cover >= 50 % of the best-covered track (avoid edge slivers)
        cover = max(v[2] for v in pt.values())
        cur_trk, (d, p, _) = max(((t, v) for t, v in pt.items() if v[2] >= 0.5 * cover), key=lambda kv: kv[1][0])
        table.append(dict(osm_id=e["osm_id"], commune=e["commune"], province=e["province"], ref_date=ref,

@router.get("/choropleth/composite")
def choropleth_composite(date: str, track: Track):
    """Playback companion of the mosaic: per commune, its latest observation on or before (date, track) that covers
    >= 50 % of the commune's best-covered pass and is at most max_age days old. Returns {osm_id: {pct, age, date, track}}."""
    if not any(f["date"] == date and f["track"] == track for f in store().manifest["frames"]):
        raise HTTPException(404, "Scene not found")
    now = dt.date.fromisoformat(date)
    max_age = store().composite_max_age
    out = {}
    for oid, rows in store().series.items():
        best_valid = max((r["valid_km2"] for r in rows), default=0)
        eligible = [r for r in rows if (r["date"], r["track"]) <= (date, track) and r["valid_km2"] >= 0.5 * best_valid]
        if not eligible:
            continue
        r = max(eligible, key=lambda x: (x["date"], x["track"]))
        age = (now - dt.date.fromisoformat(r["date"])).days
        if age <= max_age:
            out[str(oid)] = dict(pct=r["water_pct"], age=age, date=r["date"], track=r["track"])
    return out

def _num(v):
    if v in (None, "", "None"):
        return None
    try:
        f = float(v)
        return int(f) if f.is_integer() and "." not in str(v) else f
    except ValueError:
        return {"True": True, "False": False}.get(v, v)


def _csv(name):
    p = DATA_DIR / "stats" / name
    if not p.exists():
        return []
    with open(p, newline="", encoding="utf-8") as fh:
        return [{k: _num(v) for k, v in r.items()} for r in csv.DictReader(fh)]


def _sowing_row(r):
    """Turn 'latest_dates' / 'water_pct_by_track' strings into a structured per-track list."""
    dates = dict(x.split(":", 1) for x in str(r.pop("latest_dates", "")).split(";") if ":" in x)
    pcts = dict(x.split(":", 1) for x in str(r.pop("water_pct_by_track", "")).split(";") if ":" in x)
    r["per_track"] = [{"track": t, "date": dates.get(t), "water_pct": float(pcts[t])} for t in sorted(pcts)]
    return r

"""Per-date water maps, flood-timing rasters and commune statistics.

    python -m skyradar_nisar.timing
Outputs (data/processed):
  water/<date>_<track>_water.tif        uint8 0 dry, 1 water, 255 no data (permanent water excluded)
  summary/{flood_count,first_flood_day,recession_day}.tif   days since 2026-06-01; 9999 = still wet at last image
  stats/commune_water_by_date.csv       one row per commune x date x track (keep tracks separate!)
  stats/commune_flood_timing.csv        per commune: peak, first/last date >= 20 %, halved-after-peak date
"""
import csv, json, datetime as dt
import numpy as np, rasterio
from rasterio.features import rasterize
from shapely.geometry import shape
from shapely.ops import transform as stransform
from pyproj import Transformer
from . import config as C
from .grid import Grid, read_db
from .water import water_map, permanent_water

D0 = dt.date(2026, 6, 1)


def _rd(p):
    with rasterio.open(p) as s:
        return s.read(1)


def commune_zones(grid):
    fc = json.loads((C.DATA / "boundaries" / "communes_aoi.geojson").read_text(encoding="utf-8"))
    to = Transformer.from_crs(4326, grid.epsg, always_xy=True).transform
    zone = rasterize([(stransform(to, shape(f["geometry"])), i + 1) for i, f in enumerate(fc["features"])],
                     out_shape=grid.shape, transform=grid.transform, fill=0, dtype="int32")
    return zone, [f["properties"] for f in fc["features"]]


def main():
    P = C.PROCESSED; grid = Grid()
    for d in ["water", "summary", "stats"]:
        (P / d).mkdir(parents=True, exist_ok=True)
    qa = json.loads((P / "qa.json").read_text(encoding="utf-8")); keys = sorted(qa)
    s1 = [P / f"dswx_s1/{k}_WTR.tif" for k in keys if (P / f"dswx_s1/{k}_WTR.tif").exists()]
    perm = permanent_water([_rd(p) for p in s1]) if s1 else np.zeros(grid.shape, bool)
    zone, props = commune_zones(grid); nz = len(props) + 1
    maps, rows = {}, []
    for k in keys:
        w, valid = water_map(read_db(P / f"gcov/{k}_HH.tif"), read_db(P / f"gcov/{k}_HV.tif"))
        w &= ~perm
        m = np.where(valid, w.astype(np.uint8), 255).astype(np.uint8)
        with rasterio.open(P / f"water/{k}_water.tif", "w", **grid.profile("uint8", 255)) as dst:
            dst.write(m, 1)
        maps[k] = m
        v = np.bincount(zone[m != 255], minlength=nz); ww = np.bincount(zone[m == 1], minlength=nz)
        for z, pr in enumerate(props, start=1):
            if v[z] * C.RES ** 2 / 1e6 >= 3:
                rows.append(dict(date=k[:10], track=k[11:], maturity=qa[k]["maturity"], osm_id=pr["osm_id"], commune=pr["name"],
                                 province=pr["province"], valid_km2=round(v[z] * 4e-4, 2), water_km2=round(ww[z] * 4e-4, 2),
                                 water_pct=round(ww[z] / v[z] * 100, 1)))
    _write(P / "stats/commune_water_by_date.csv", rows)

    # flood timing over the season, all tracks merged by date (a water/dry map is geometry-independent)
    season = [k for k in keys if k[:10] >= C.FLOOD_SEASON_START]
    days = np.array([(dt.date.fromisoformat(k[:10]) - D0).days for k in season], np.int16)[:, None, None]
    st = np.stack([maps[k] for k in season]); obs = st != 255; wet = st == 1
    ever = wet.any(0); BIG = 32767
    first = np.where(wet, days, BIG).min(0); last = np.where(wet, days, -1).max(0)
    reces = np.where(obs & ~wet & (days > last[None]), days, BIG).min(0)
    for name, arr in [("flood_count", np.where(obs.any(0), wet.sum(0), 255)), ("first_flood_day", np.where(ever, first, -1)),
                      ("recession_day", np.where(ever, np.where(reces < BIG, reces, 9999), -1))]:
        dtype, nod = ("uint8", 255) if name == "flood_count" else ("int16", -1)
        with rasterio.open(P / f"summary/{name}.tif", "w", **grid.profile(dtype, nod)) as dst:
            dst.write(arr.astype(dtype), 1)

    # per-commune timing table (per track to avoid mixing viewing geometries; 047D first if present)
    tab = []
    by = {}
    for r in rows:
        by.setdefault((r["osm_id"], r["track"]), []).append(r)
    for (oid, trk), rs in by.items():
        rs.sort(key=lambda r: r["date"])
        seas = [r for r in rs if r["date"] >= C.FLOOD_SEASON_START and r["valid_km2"] >= 5]
        if not seas:
            continue
        pk = max(seas, key=lambda r: r["water_pct"])
        above = [r for r in seas if r["water_pct"] >= 20]
        halved = next((r["date"] for r in seas if r["date"] > pk["date"] and r["water_pct"] < pk["water_pct"] / 2), None)
        tab.append(dict(osm_id=oid, commune=rs[0]["commune"], province=rs[0]["province"], track=trk,
                        peak_pct=pk["water_pct"], peak_date=pk["date"], first_date_ge20pct=above[0]["date"] if above else None,
                        last_date_ge20pct=above[-1]["date"] if above else None, halved_after_peak=halved,
                        last_date=seas[-1]["date"], last_pct=seas[-1]["water_pct"]))
    _write(P / "stats/commune_flood_timing.csv", tab)

import type { SummaryLayer, Track } from './data/types.ts';

export type Tab = 'overview' | 'map' | 'reports' | 'validation' | 'about';
export type TrackFilter = 'all' | Track;
export type SummaryId = SummaryLayer['id'] | null;

export interface LayerToggles {
  basemap: boolean;
  radar: boolean;
  water: boolean;
  communes: boolean;
  provinces: boolean;
}

export interface Opacities {
  radar: number;
  water: number;
  communes: number;
}

export interface State {
  tab: Tab;
  mapMode: 'latest' | 'scene';
  trackFilter: TrackFilter;
  selectedDate: string;
  selectedTrack: Track;
  selectedCommune: string | null;
  /** Incremented to ask the map to fly to the selected commune. */
  flyNonce: number;
  activeSummary: SummaryId;
  layers: LayerToggles;
  opacity: Opacities;
  playing: boolean;
  radarMissing: boolean;
}

function restoredSnapshot(): Partial<State> {
  try {
    const raw = sessionStorage.getItem('waterpulse.snapshot');
    if (!raw) return {};
    sessionStorage.removeItem('waterpulse.snapshot');
    return JSON.parse(raw) as Partial<State>;
  } catch {
    return {};
  }
}

export const state: State = {
  tab: 'overview',
  mapMode: 'latest',
  trackFilter: 'all',
  selectedDate: '',
  selectedTrack: '148D',
  selectedCommune: null,
  flyNonce: 0,
  activeSummary: null,
  layers: { basemap: true, radar: true, water: true, communes: true, provinces: true },
  opacity: { radar: 0.85, water: 0.9, communes: 0.6 },
  playing: false,
  radarMissing: false,
  ...restoredSnapshot(),
};

type Listener = (s: State, prev: State) => void;
const listeners = new Set<Listener>();

export function subscribe(fn: Listener): () => void {
  listeners.add(fn);
  return () => listeners.delete(fn);
}

export function set(patch: Partial<State>): void {
  const prev = { ...state };
  let changed = false;
  for (const key of Object.keys(patch) as (keyof State)[]) {
    if (state[key] !== patch[key]) changed = true;
  }
  if (!changed) return;
  Object.assign(state, patch);
  listeners.forEach((fn) => fn(state, prev));
}

const SOWING = '2026-11-01';
const GAP_DAYS = 15;

export function mountTimeline(host: HTMLElement, scenes: Scene[]): void {
  const all = [...scenes].sort((a, b) => a.date.localeCompare(b.date) || a.track.localeCompare(b.track));
  const minMs = toMs(all[0].date);
  const maxMs = Math.max(toMs(all[all.length - 1].date), toMs(SOWING)) + 4 * 86400000;
  const xPct = (date: string) => ((toMs(date) - minMs) / (maxMs - minMs)) * 100;

  const visible = (f: TrackFilter) => (f === 'all' ? all : all.filter((s) => s.track === f));
  const same = (a: Scene, b: { date: string; track: Track }) => a.date === b.date && a.track === b.track;
  const current = (s: State) => all.find((x) => same(x, { date: s.selectedDate, track: s.selectedTrack }));

  function select(scene: Scene) {
    set({ selectedDate: scene.date, selectedTrack: scene.track, mapMode: 'scene' });
  }

  function setFilter(f: TrackFilter) {
    if (f === state.trackFilter) return;
    set({ trackFilter: f });
    if (f !== 'all' && state.selectedTrack !== f) {
      const pool = visible(f);
      const target = toMs(state.selectedDate);
      const nearest = pool.reduce((best, s) => (Math.abs(toMs(s.date) - target) < Math.abs(toMs(best.date) - target) ? s : best), pool[0]);
      select(nearest);
    }
  }

  function step(delta: number): boolean {
    const list = visible(state.trackFilter);
    const idx = list.findIndex((s) => same(s, { date: state.selectedDate, track: state.selectedTrack }));
    const next = list[idx + delta];
    if (!next) return false;
    select(next);
    return true;
  }

  // ---- play ----
  let timer: number | undefined;
  function stop() {
    if (timer !== undefined) window.clearInterval(timer);
    timer = undefined;
    if (state.playing) set({ playing: false });
  }
  function play() {
    const list = visible(state.trackFilter);
    const idx = list.findIndex((s) => same(s, { date: state.selectedDate, track: state.selectedTrack }));
    // From the "latest per commune" view, or at the end, start from the first acquisition (select() switches to one-acquisition mode).
    if (state.mapMode === 'latest' || idx < 0 || idx >= list.length - 1) select(list[0]);
    set({ playing: true });
    timer = window.setInterval(() => {
      if (!step(1)) stop();
    }, PLAY_INTERVAL_MS);
  }

  // ---- DOM ----
  const playBtn = h('button', { class: 'tl-play', type: 'button', 'aria-label': T.timeline.play });
  const dateEl = h('div', { class: 'tl-date' });
  const trackEl = h('span', { class: 'tl-track' });
  const maturityEl = h('span', { class: 'maturity' });
  const infoEl = h('div', { class: 'tl-info' });
  const summaryNote = h('div', { class: 'tl-summary-note', text: T.timeline.summaryNote });
  const latestNote = h('div', { class: 'tl-latest-note', text: T.timeline.latestNote });

  const seg = h('div', { class: 'tl-seg', role: 'group', 'aria-label': T.timeline.trackGroup, title: T.timeline.trackHelp });
  const segBtns = new Map<TrackFilter, HTMLButtonElement>();
  for (const f of ['all', ...TRACKS] as TrackFilter[]) {
    const b = h('button', { type: 'button', text: f === 'all' ? T.timeline.all : f });
    if (f !== 'all') b.style.setProperty('--dot', TRACK_COLORS[f]);
    b.addEventListener('click', () => setFilter(f));
    segBtns.set(f, b);
    seg.append(b);
  }
  const collapseBtn = h('button', { class: 'tl-collapse', type: 'button', 'aria-label': T.timeline.collapse, title: T.timeline.collapse }, '▾');

  const head = h(
    'div',
    { class: 'tl-head' },
    playBtn,
    h('div', { class: 'tl-now' }, dateEl, h('div', { class: 'tl-meta' }, trackEl, maturityEl)),
    infoEl,
    h('div', { class: 'tl-spacer' }),
    seg,
    collapseBtn,
  );

  const plot = h('div', { class: 'tl-plot' });
  const axis = h('div', { class: 'tl-axis' }, plot);
  const legend = h(
    'div',
    { class: 'tl-legend' },
    h('span', {}, h('i', { class: 'tl-key filled' }), T.timeline.validated),
    h('span', {}, h('i', { class: 'tl-key hollow' }), T.timeline.provisional),
    h('span', { class: 'tl-legend-note', text: T.timeline.note }),
  );
  const body = h('div', { class: 'tl-body' }, axis, legend);
  host.classList.add('timeline');
  if (window.innerWidth < 768) {
    host.classList.add('collapsed');
    collapseBtn.textContent = '▴';
  }
  host.append(head, summaryNote, latestNote, body);

  const dots = new Map<string, HTMLButtonElement>();
  const key = (s: { date: string; track: Track }) => `${s.date}|${s.track}`;
  const playhead = h('div', { class: 'tl-playhead' });

  function buildPlot(filter: TrackFilter) {
    plot.replaceChildren();
    dots.clear();
    const grid = h('div', { class: 'tl-grid' });
    const sow = h('div', { class: 'tl-sow', style: `left:${xPct(SOWING)}%` });
    sow.append(h('span', { text: T.timeline.sowing }));
    grid.append(sow);
    for (let m = 6; m <= 10; m++) {
      const d = `2026-${String(m + 1).padStart(2, '0')}-01`;
      if (toMs(d) > minMs && toMs(d) < maxMs && d !== SOWING) grid.append(h('div', { class: 'tl-month', style: `left:${xPct(d)}%` }));
    }
    plot.append(grid);

    const lanes = filter === 'all' ? TRACKS : [filter as Track];
    for (const t of lanes) {
      const lane = h('div', { class: 'tl-lane' }, h('span', { class: 'tl-lane-label', text: t, style: `color:${TRACK_COLORS[t]}` }));
      const list = all.filter((s) => s.track === t);
      list.forEach((s, i) => {
        const prev = list[i - 1];
        if (prev && daysBetween(prev.date, s.date) > GAP_DAYS) {
          const gap = h('div', { class: 'tl-gap', style: `left:${xPct(prev.date)}%;width:${xPct(s.date) - xPct(prev.date)}%` }, h('span', { text: T.timeline.gap(daysBetween(prev.date, s.date)) }));
          lane.append(gap);
        }
      });
      for (const s of list) {
        const dot = h('button', {
          class: `tl-dot ${s.maturity}`,
          type: 'button',
          style: `left:${xPct(s.date)}%;--c:${TRACK_COLORS[s.track]}`,
          'aria-label': T.timeline.dotLabel(fmtDate(s.date), s.track, MATURITY_LABEL(s.maturity)),
          title: `${fmtDate(s.date)} · ${s.track} · ${MATURITY_LABEL(s.maturity)}`,
        });
        dot.addEventListener('click', () => {
          stop();
          select(s);
        });
        dots.set(key(s), dot);
        lane.append(dot);
      }
      plot.append(lane);
    }
    plot.append(playhead);

    const ticks = h('div', { class: 'tl-ticks' });
    for (const m of [6, 7, 8, 9, 10] as const) {
      const label = T.months[m];
      const d = `2026-${String(m + 1).padStart(2, '0')}-01`;
      if (toMs(d) < maxMs) ticks.append(h('span', { style: `left:${xPct(d)}%`, text: label }));
    }
    ticks.append(h('span', { class: 'tl-tick-start', style: `left:0%`, text: fmtShort(all[0].date) }));
    plot.append(ticks);
  }

  // scrubbing: pointer position -> nearest real acquisition date (never an interpolated one)
  let scrubbing = false;
  function scrub(ev: PointerEvent) {
    const rect = plot.getBoundingClientRect();
    const frac = Math.min(1, Math.max(0, (ev.clientX - rect.left) / rect.width));
    const ms = minMs + frac * (maxMs - minMs);
    const list = visible(state.trackFilter);
    const nearest = list.reduce((best, s) => (Math.abs(toMs(s.date) - ms) < Math.abs(toMs(best.date) - ms) ? s : best), list[0]);
    if (!same(nearest, { date: state.selectedDate, track: state.selectedTrack })) select(nearest);
  }
  plot.addEventListener('pointerdown', (ev) => {
    if (state.activeSummary) return;
    scrubbing = true;
    stop();
    plot.setPointerCapture(ev.pointerId);
    scrub(ev);
  });
  plot.addEventListener('pointermove', (ev) => {
    if (scrubbing) scrub(ev);
  });
  const endScrub = () => (scrubbing = false);
  plot.addEventListener('pointerup', endScrub);
  plot.addEventListener('pointercancel', endScrub);

  playBtn.addEventListener('click', () => (state.playing ? stop() : play()));
  collapseBtn.addEventListener('click', () => {
    const c = host.classList.toggle('collapsed');
    collapseBtn.textContent = c ? '▴' : '▾';
    window.dispatchEvent(new Event('resize'));
  });
  document.addEventListener('keydown', (ev) => {
    if (state.tab !== 'map' || state.activeSummary) return;
    const t = ev.target as HTMLElement;
    if (/^(INPUT|SELECT|TEXTAREA)$/.test(t.tagName) && (t as HTMLInputElement).type !== 'checkbox' && (t as HTMLInputElement).type !== 'radio') return;
    if (ev.key === 'ArrowRight' || ev.key === 'ArrowLeft') {
      ev.preventDefault();
      stop();
      step(ev.key === 'ArrowRight' ? 1 : -1);
    }
  });

  // ---- render ----
  let builtFor: TrackFilter | null = null;
  function render(s: State) {
    if (builtFor !== s.trackFilter) {
      buildPlot(s.trackFilter);
      builtFor = s.trackFilter;
    }
    const scene = current(s);
    for (const [f, b] of segBtns) b.setAttribute('aria-pressed', String(f === s.trackFilter));
    host.classList.toggle('summary-mode', s.activeSummary !== null);
    host.classList.toggle('latest-mode', s.mapMode === 'latest' && s.activeSummary === null);
    playBtn.textContent = s.playing ? '❚❚' : '▶';
    playBtn.setAttribute('aria-label', s.playing ? T.timeline.pause : T.timeline.play);
    if (!scene) return;
    dateEl.textContent = fmtDate(scene.date);
    trackEl.textContent = T.timeline.trackLabel(scene.track);
    trackEl.style.setProperty('--c', TRACK_COLORS[scene.track]);
    maturityEl.textContent = MATURITY_LABEL(scene.maturity);
    maturityEl.className = `maturity ${scene.maturity}`;
    const prev = [...all].reverse().find((x) => x.track === scene.track && x.date < scene.date);
    infoEl.textContent = prev
      ? T.timeline.daysAfter(daysBetween(prev.date, scene.date))
      : T.timeline.firstImage;
    const idx = visible(s.trackFilter).findIndex((x) => same(x, scene));
    infoEl.title = T.timeline.imageOf(idx + 1, visible(s.trackFilter).length);
    for (const [k, d] of dots) d.classList.toggle('selected', k === key(scene));
    playhead.style.left = `${xPct(scene.date)}%`;
  }

  subscribe((s) => {
    // Other controls pause by setting playing=false or switching to 'latest'; clear the timer too, or it keeps stepping.
    if ((s.mapMode === 'latest' || !s.playing) && timer !== undefined) stop();
    render(s);
  });
  render(state);
}

import { T } from '../i18n.ts';
import { heroBlock } from './about.ts';
import { h } from './dom.ts';
import { openGuide } from './guide.ts';

const KEY = 'waterpulse.hero.seen';

function seen(): boolean {
  try {
    return localStorage.getItem(KEY) === '1';
  } catch {
    return false;
  }
}

function remember(): void {
  try {
    localStorage.setItem(KEY, '1');
  } catch {
    /* storage can be blocked (private mode); the overlay then simply shows again */
  }
}

/** First-visit overlay with the key message. Closable; remembered in localStorage. */
export function maybeShowHero(): void {
  if (seen() || location.hash === '#about') return;
  const overlay = h('div', { class: 'hero-overlay', role: 'dialog', 'aria-modal': 'true', 'aria-label': T.hero.welcome });
  const close = () => {
    remember();
    overlay.remove();
    document.removeEventListener('keydown', onKey);
  };
  const onKey = (ev: KeyboardEvent) => {
    if (ev.key === 'Escape') close();
  };
  const howTo = h('button', { class: 'link', type: 'button', text: T.hero.howTo });
  howTo.addEventListener('click', () => {
    close();
    openGuide();
  });
  const skip = h('button', { class: 'link', type: 'button', text: T.hero.skip });
  skip.addEventListener('click', close);
  overlay.append(heroBlock(close, h('span', { class: 'hero-links' }, howTo, skip)));
  overlay.addEventListener('pointerdown', (ev) => {
    if (ev.target === overlay) close();
  });
  document.addEventListener('keydown', onKey);
  document.body.append(overlay);
  overlay.querySelector<HTMLElement>('.btn-primary')?.focus();
}

async function buildMapView(view: HTMLElement, menuBtn: HTMLElement) {
  const [scenes, summaries] = await Promise.all([getScenes(), getSummaryLayers()]);
  if (!scenes.some(s => s.date === state.selectedDate && s.track === state.selectedTrack)) {
    const latest = [...scenes].sort((a, b) => a.date.localeCompare(b.date)).at(-1);
    if (latest) set({ selectedDate: latest.date, selectedTrack: latest.track });
  }
  if (state.activeSummary && !summaries.some((l) => l.id === state.activeSummary)) set({ activeSummary: null });

  const sidebar = h('aside', { 'aria-label': T.app.menu });
  const context = h('section');
  const mapEl = h('div', { id: 'map' });
  const note = h('div', { class: 'map-note', role: 'status', text: T.map.radarMissing });
  const hint = h('div', { class: 'map-hint', text: T.map.hint });
  const mapWrap = h('div', { class: 'map-wrap' }, mapEl, note, hint);
  const timeline = h('div', { 'aria-label': T.timeline.region });
  const panel = h('aside', {});
  mapWrap.append(panel);
  view.append(context, sidebar, mapWrap, timeline);

  mountSidebar(sidebar, summaries);
  mountTimeline(timeline, scenes);
  mountCommunePanel(panel, scenes);
  mountMap(mapEl);
  await mountMapContext(context, scenes);

  menuBtn.addEventListener('click', () => sidebar.classList.toggle('open'));
  mapWrap.addEventListener('pointerdown', () => sidebar.classList.remove('open'));

  const sync = () => {
    view.classList.toggle('latest-mode', state.mapMode === 'latest');
    note.classList.toggle('show', state.mapMode === 'scene' && state.radarMissing && state.layers.radar && state.activeSummary === null);
    hint.classList.toggle('show', state.selectedCommune === null);
  };
  subscribe(sync);
  sync();
}

    });
}

export type Track = '047D' | '083A' | '148D';
export type Maturity = 'provisional' | 'validated';
export type Bounds = [number, number, number, number];

export interface Scene {

import type { CommuneCollection, CommuneProps, Scene } from './types.ts';

/** [Codex] An assessment needs sufficient coverage; missing is never zero water. */
export function hasAssessment(p: CommuneProps): boolean {
  return p.coverage_sufficient === true && p.last_pct !== null && p.last_date !== null;
}

export function latestValues(collection: CommuneCollection): Record<string, number> {
  return Object.fromEntries(collection.features.filter(f => hasAssessment(f.properties)).map(f => [f.properties.osm_id, f.properties.last_pct!]));
}

export function referenceScene(p: CommuneProps, scenes: Scene[]): Scene | undefined {
  return scenes.find(s => s.date === p.last_date && s.track === p.reference_track);
}

function median(values: number[]): number {
  const s = [...values].sort((a, b) => a - b);
  const m = s.length >> 1;
  return s.length % 2 ? s[m] : (s[m - 1] + s[m]) / 2;
}

const range = (v: number[]) => `${fmt2(Math.min(...v))}–${fmt2(Math.max(...v))}`;

function hlsTable(rows: ValidationRow[]): HTMLElement {
  type Pair = { label: string; a: keyof ValidationRow; b: keyof ValidationRow };
  const c = T.validation.hlsCols;
  const pairs: Pair[] = [
    { label: c.f1, a: 'f1_nisar', b: 'f1_s1' },
    { label: c.recall, a: 'recall_nisar', b: 'recall_s1' },
    { label: c.precision, a: 'precision_nisar', b: 'precision_s1' },
  ];
  const head1 = h('tr', {}, h('th', { rowspan: '2', text: c.date }), ...pairs.map((p) => h('th', { colspan: '2', class: 'grp', text: p.label })), h('th', { rowspan: '2', text: c.only }));
  const head2 = h('tr', {}, ...pairs.flatMap(() => [h('th', { class: 'sub', text: 'NISAR' }), h('th', { class: 'sub', text: 'S-1' })]));
  const body = rows.map((r) =>
    h(
      'tr',
      {},
      h('td', { text: fmtShort(r.date) }),
      ...pairs.flatMap((p) => {
        const a = r[p.a] as number | null;

        return [h('td', { class: `num ${a !== null && b !== null && a > b ? 'best' : ''}`, text: a === null ? '—' : fmt2(a) }), h('td', { class: `num ${a !== null && b !== null && b > a ? 'best' : ''}`, text: b === null ? '—' : fmt2(b) })];

    ),
  );
  return h('div', { class: 'ct-wrap' }, h('table', { class: 'ct vt' }, h('thead', {}, head1, head2), h('tbody', {}, ...body)));
}

import { spawnSync } from 'node:child_process';
import { fileURLToPath } from 'node:url';
import path from 'node:path';

const root = fileURLToPath(new URL('..', import.meta.url));
const repo = path.resolve(process.env.NISAR_REPO ?? path.join(root, '../skyradar-nisar'));
const python = process.env.PYTHON ?? 'python3';
function run(command, args, env = process.env) {
  // npm is a .cmd shim on Windows and cannot be spawned without a shell.
  const result = spawnSync(command, args, { cwd: root, env, stdio: 'inherit', shell: command === 'npm' && process.platform === 'win32' });
  if (result.error) throw result.error;
  if (result.status !== 0) process.exit(result.status ?? 1);
}
