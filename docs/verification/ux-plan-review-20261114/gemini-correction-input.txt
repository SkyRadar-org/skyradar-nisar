import type { SummaryLayer, Track } from './data/types.ts';

export type Tab = 'overview' | 'map' | 'reports' | 'validation' | 'about';
/** 'pro': radar + age-faded water + per-track timeline; 'simple': one-colour water and one timeline bar (needs blended frames). */
export type ViewMode = 'pro' | 'simple';
export type TrackFilter = 'all' | Track;
export type SummaryId = SummaryLayer['id'] | null;

export interface LayerToggles {
  basemap: boolean;
  radar: boolean;
  water: boolean;
  communes: boolean;
  provinces: boolean;
}

export interface Opacities {
  radar: number;
  water: number;
  communes: number;
}

export interface State {
  tab: Tab;
  mapMode: 'latest' | 'scene';
  trackFilter: TrackFilter;
  selectedDate: string;
  selectedTrack: Track;
  selectedCommune: string | null;
  /** Incremented to ask the map to fly to the selected commune. */
  flyNonce: number;
  activeSummary: SummaryId;
  layers: LayerToggles;
  opacity: Opacities;
  viewMode: ViewMode;
  /** The last selection jumped over a long gap without images: swap the map instantly instead of cross-fading. */
  cut: boolean;
  playing: boolean;
  radarMissing: boolean;
}

function restoredSnapshot(): Partial<State> {
  try {
    const raw = sessionStorage.getItem('waterpulse.snapshot');
    if (!raw) return {};
    sessionStorage.removeItem('waterpulse.snapshot');
    return JSON.parse(raw) as Partial<State>;
  } catch {
    return {};
  }
}

export const state: State = {
  tab: 'overview',
  mapMode: 'latest',
  trackFilter: 'all',
  selectedDate: '',
  selectedTrack: '148D',
  selectedCommune: null,
  flyNonce: 0,
  activeSummary: null,
  layers: { basemap: true, radar: true, water: true, communes: true, provinces: true },
  opacity: { radar: 0.85, water: 0.9, communes: 0.6 },
  playing: false,
  viewMode: 'pro',
  cut: false,
  radarMissing: false,
  ...restoredSnapshot(),
};

type Listener = (s: State, prev: State) => void;
const listeners = new Set<Listener>();

export function subscribe(fn: Listener): () => void {
  listeners.add(fn);
  return () => listeners.delete(fn);
}

export function set(patch: Partial<State>): void {
  const prev = { ...state };
  let changed = false;
  for (const key of Object.keys(patch) as (keyof State)[]) {
    if (state[key] !== patch[key]) changed = true;
  }
  if (!changed) return;
  Object.assign(state, patch);
  listeners.forEach((fn) => fn(state, prev));
}

import { PLAY_INTERVAL_MS } from '../config.ts';
import type { Scene, Track } from '../data/types.ts';
import { daysBetween, fmtDate, fmtShort, toMs } from '../format.ts';
import { T } from '../i18n.ts';

import { set, state, subscribe, type State, type TrackFilter, type ViewMode } from '../state.ts';
import { h } from './dom.ts';

const TRACKS: Track[] = ['047D', '083A', '148D'];
const SOWING = '2026-11-01';
const GAP_DAYS = 15;
const SIMPLE_COLOR = '#1d4ed8';

export function mountTimeline(host: HTMLElement, scenes: Scene[]): void {
  const all = [...scenes].sort((a, b) => a.date.localeCompare(b.date) || a.track.localeCompare(b.track));
  const minMs = toMs(all[0].date);
  const maxMs = Math.max(toMs(all[all.length - 1].date), toMs(SOWING)) + 4 * 86400000;
  const xPct = (date: string) => ((toMs(date) - minMs) / (maxMs - minMs)) * 100;

  const visible = (f: TrackFilter) => (f === 'all' ? all : all.filter((s) => s.track === f));
  const same = (a: Scene, b: { date: string; track: Track }) => a.date === b.date && a.track === b.track;
  const current = (s: State) => all.find((x) => same(x, { date: s.selectedDate, track: s.selectedTrack }));

  function select(scene: Scene) {
    // A jump across a long stretch without images is shown as a cut, not a fade between unrelated frames.
    const from = current(state);
    const cut = !!from && Math.abs(daysBetween(from.date, scene.date)) > GAP_DAYS;
    set({ selectedDate: scene.date, selectedTrack: scene.track, mapMode: 'scene', cut });
  }

  function setFilter(f: TrackFilter) {
    if (f === state.trackFilter) return;
    set({ trackFilter: f });
    if (f !== 'all' && state.selectedTrack !== f) {
      const pool = visible(f);
      const target = toMs(state.selectedDate);
      const nearest = pool.reduce((best, s) => (Math.abs(toMs(s.date) - target) < Math.abs(toMs(best.date) - target) ? s : best), pool[0]);
      select(nearest);
    }
  }

  function step(delta: number): boolean {
    const list = visible(state.trackFilter);
    const idx = list.findIndex((s) => same(s, { date: state.selectedDate, track: state.selectedTrack }));
    const next = list[idx + delta];
    if (!next) return false;
    select(next);
    return true;
  }

  // ---- play ----
  // Blended frames are paced by the real time between acquisitions (so the season "breathes" at a steady rate);
  // a gap without images is skipped quickly (the map cuts instead of fading). Raw frames keep the fixed interval.
  let timer: number | undefined;
  function stop() {
    if (timer !== undefined) window.clearTimeout(timer);
    timer = undefined;
    if (state.playing) set({ playing: false });
  }
  function nextDelay(): number {
    const sc = current(state);
    const nxt = visible(state.trackFilter)[visible(state.trackFilter).findIndex((x) => sc && same(x, sc)) + 1];
    if (!sc || !nxt || !sc.composite) return PLAY_INTERVAL_MS;
    const days = daysBetween(sc.date, nxt.date);
    return days > GAP_DAYS ? 500 : Math.min(1500, Math.max(650, days * 150));
  }
  function tick() {
    timer = window.setTimeout(() => {
      if (!step(1)) return stop();
      tick();
    }, nextDelay());
  }
  function play() {
    const list = visible(state.trackFilter);
    const idx = list.findIndex((s) => same(s, { date: state.selectedDate, track: state.selectedTrack }));
    // From the "latest per commune" view, or at the end, start from the first acquisition (select() switches to one-acquisition mode).
    if (state.mapMode === 'latest' || idx < 0 || idx >= list.length - 1) select(list[0]);
    set({ playing: true });
    tick();
  }

  // ---- DOM ----
  const playBtn = h('button', { class: 'tl-play', type: 'button', 'aria-label': T.timeline.play });
  const dateEl = h('div', { class: 'tl-date' });
  const trackEl = h('span', { class: 'tl-track' });
  const maturityEl = h('span', { class: 'maturity', title: T.timeline.maturityHelp });
  const infoEl = h('div', { class: 'tl-info' });
  const summaryNote = h('div', { class: 'tl-summary-note', text: T.timeline.summaryNote });
  const latestNote = h('div', { class: 'tl-latest-note', text: T.timeline.latestNote });

  const seg = h('div', { class: 'tl-seg', role: 'group', 'aria-label': T.timeline.trackGroup, title: T.timeline.trackHelp });
  const segBtns = new Map<TrackFilter, HTMLButtonElement>();
  for (const f of ['all', ...TRACKS] as TrackFilter[]) {
    const b = h('button', { type: 'button', text: f === 'all' ? T.timeline.all : f });
    if (f !== 'all') b.style.setProperty('--dot', TRACK_COLORS[f]);
    b.addEventListener('click', () => setFilter(f));
    segBtns.set(f, b);
    seg.append(b);
  }
  const collapseBtn = h('button', { class: 'tl-collapse', type: 'button', 'aria-label': T.timeline.collapse, title: T.timeline.collapse }, '▾');
  const hasMosaic = scenes.some((s) => s.composite);
  const viewSeg = h('div', { class: 'tl-seg tl-view', role: 'group', 'aria-label': T.timeline.modeGroup, title: T.timeline.modeHelp });
  const viewBtns = new Map<ViewMode, HTMLButtonElement>();
  for (const m of ['pro', 'simple'] as ViewMode[]) {
    const b = h('button', { type: 'button', text: m === 'pro' ? T.timeline.modePro : T.timeline.modeSimple });
    b.addEventListener('click', () => {
      if (m === state.viewMode) return;
      // The simple view is one timeline bar over every date, so orbit filtering does not apply.
      set(m === 'simple' ? { viewMode: m, trackFilter: 'all', cut: false } : { viewMode: m, cut: false });
    });
    viewBtns.set(m, b);
    viewSeg.append(b);
  }
  viewSeg.hidden = !hasMosaic;

  const head = h(
    'div',
    { class: 'tl-head' },
    playBtn,
    h('div', { class: 'tl-now' }, dateEl, h('div', { class: 'tl-meta' }, trackEl, maturityEl)),
    infoEl,
    h('div', { class: 'tl-spacer' }),
    viewSeg,
    seg,
    collapseBtn,
  );

  const plot = h('div', { class: 'tl-plot' });
  const axis = h('div', { class: 'tl-axis' }, plot);
  const legend = h(
    'div',
    { class: 'tl-legend' },
    h('span', { title: T.timeline.maturityHelp }, h('i', { class: 'tl-key filled' }), T.timeline.validated),
    h('span', { title: T.timeline.maturityHelp }, h('i', { class: 'tl-key hollow' }), T.timeline.provisional),
    h('span', { class: 'tl-legend-note', text: T.timeline.note }),
  );
  const body = h('div', { class: 'tl-body' }, axis, legend);
  host.classList.add('timeline');
  if (window.innerWidth < 768) {
    host.classList.add('collapsed');
    collapseBtn.textContent = '▴';
  }
  host.append(head, summaryNote, latestNote, body);

  const dots = new Map<string, HTMLButtonElement>();
  const key = (s: { date: string; track: Track }) => `${s.date}|${s.track}`;
  const playhead = h('div', { class: 'tl-playhead' });

  function buildPlot(filter: TrackFilter, mode: ViewMode) {
    plot.replaceChildren();
    dots.clear();
    const grid = h('div', { class: 'tl-grid' });
    const sow = h('div', { class: 'tl-sow', style: `left:${xPct(SOWING)}%` });
    sow.append(h('span', { text: T.timeline.sowing }));
    grid.append(sow);
    for (let m = 6; m <= 10; m++) {
      const d = `2026-${String(m + 1).padStart(2, '0')}-01`;
      if (toMs(d) > minMs && toMs(d) < maxMs && d !== SOWING) grid.append(h('div', { class: 'tl-month', style: `left:${xPct(d)}%` }));
    }
    plot.append(grid);

    const simple = mode === 'simple';
    const lanes: (Track | null)[] = simple ? [null] : filter === 'all' ? TRACKS : [filter as Track];
    for (const t of lanes) {
      const lane = h('div', { class: 'tl-lane' });
      if (t) lane.append(h('span', { class: 'tl-lane-label', text: t, style: `color:${TRACK_COLORS[t]}` }));
      const list = t ? all.filter((s) => s.track === t) : all;
      list.forEach((s, i) => {
        const prev = list[i - 1];
        if (prev && daysBetween(prev.date, s.date) > GAP_DAYS) {
          const gap = h('div', { class: 'tl-gap', style: `left:${xPct(prev.date)}%;width:${xPct(s.date) - xPct(prev.date)}%` }, h('span', { text: T.timeline.gap(daysBetween(prev.date, s.date)) }));
          lane.append(gap);
        }
      });
      for (const s of list) {
        const dot = h('button', {
          class: `tl-dot ${simple ? 'validated' : s.maturity}`,
          type: 'button',
          style: `left:${xPct(s.date)}%;--c:${simple ? SIMPLE_COLOR : TRACK_COLORS[s.track]}`,
          'aria-label': simple ? fmtDate(s.date) : T.timeline.dotLabel(fmtDate(s.date), s.track, MATURITY_LABEL(s.maturity)),
          title: simple ? fmtDate(s.date) : `${fmtDate(s.date)} · ${s.track} · ${MATURITY_LABEL(s.maturity)}`,
        });
        dot.addEventListener('click', () => {
          stop();
          select(s);
        });
        dots.set(key(s), dot);
        lane.append(dot);
      }
      plot.append(lane);
    }
    plot.append(playhead);

    const ticks = h('div', { class: 'tl-ticks' });
    for (const m of [6, 7, 8, 9, 10] as const) {
      const label = T.months[m];
      const d = `2026-${String(m + 1).padStart(2, '0')}-01`;
      if (toMs(d) < maxMs) ticks.append(h('span', { style: `left:${xPct(d)}%`, text: label }));
    }
    ticks.append(h('span', { class: 'tl-tick-start', style: `left:0%`, text: fmtShort(all[0].date) }));
    plot.append(ticks);
  }

  // scrubbing: pointer position -> nearest real acquisition date (never an interpolated one)
  let scrubbing = false;
  function scrub(ev: PointerEvent) {
    const rect = plot.getBoundingClientRect();
    const frac = Math.min(1, Math.max(0, (ev.clientX - rect.left) / rect.width));
    const ms = minMs + frac * (maxMs - minMs);
    const list = visible(state.trackFilter);
    const nearest = list.reduce((best, s) => (Math.abs(toMs(s.date) - ms) < Math.abs(toMs(best.date) - ms) ? s : best), list[0]);
    if (!same(nearest, { date: state.selectedDate, track: state.selectedTrack })) select(nearest);
  }
  plot.addEventListener('pointerdown', (ev) => {
    if (state.activeSummary) return;
    scrubbing = true;
    stop();
    plot.setPointerCapture(ev.pointerId);
    scrub(ev);
  });
  plot.addEventListener('pointermove', (ev) => {
    if (scrubbing) scrub(ev);
  });
  const endScrub = () => (scrubbing = false);
  plot.addEventListener('pointerup', endScrub);
  plot.addEventListener('pointercancel', endScrub);

  playBtn.addEventListener('click', () => (state.playing ? stop() : play()));
  collapseBtn.addEventListener('click', () => {
    const c = host.classList.toggle('collapsed');
    collapseBtn.textContent = c ? '▴' : '▾';
    window.dispatchEvent(new Event('resize'));
  });
  document.addEventListener('keydown', (ev) => {
    if (state.tab !== 'map' || state.activeSummary) return;
    const t = ev.target as HTMLElement;
    if (/^(INPUT|SELECT|TEXTAREA)$/.test(t.tagName) && (t as HTMLInputElement).type !== 'checkbox' && (t as HTMLInputElement).type !== 'radio') return;
    if (ev.key === 'ArrowRight' || ev.key === 'ArrowLeft') {
      ev.preventDefault();
      stop();
      step(ev.key === 'ArrowRight' ? 1 : -1);
    }
  });

  // ---- render ----
  let builtFor: string | null = null;
  function render(s: State) {
    const mode: ViewMode = hasMosaic ? s.viewMode : 'pro';
    if (builtFor !== `${s.trackFilter}|${mode}`) {
      buildPlot(s.trackFilter, mode);
      builtFor = `${s.trackFilter}|${mode}`;
    }
    host.classList.toggle('simple', mode === 'simple');
    for (const [m, b] of viewBtns) b.setAttribute('aria-pressed', String(m === mode));
    const scene = current(s);
    for (const [f, b] of segBtns) b.setAttribute('aria-pressed', String(f === s.trackFilter));
    host.classList.toggle('summary-mode', s.activeSummary !== null);
    host.classList.toggle('latest-mode', s.mapMode === 'latest' && s.activeSummary === null);
    playBtn.textContent = s.playing ? '❚❚' : '▶';
    playBtn.setAttribute('aria-label', s.playing ? T.timeline.pause : T.timeline.play);
    if (!scene) return;
    dateEl.textContent = fmtDate(scene.date);
    trackEl.textContent = T.timeline.trackLabel(scene.track);
    trackEl.style.setProperty('--c', TRACK_COLORS[scene.track]);
    maturityEl.textContent = MATURITY_LABEL(scene.maturity);
    maturityEl.className = `maturity ${scene.maturity}`;
    const prev = [...all].reverse().find((x) => x.track === scene.track && x.date < scene.date);
    infoEl.textContent = prev
      ? T.timeline.daysAfter(daysBetween(prev.date, scene.date))
      : T.timeline.firstImage;
    const idx = visible(s.trackFilter).findIndex((x) => same(x, scene));
    infoEl.title = T.timeline.imageOf(idx + 1, visible(s.trackFilter).length);
    for (const [k, d] of dots) d.classList.toggle('selected', k === key(scene));
    playhead.style.left = `${xPct(scene.date)}%`;
  }

  subscribe((s) => {
    // Other controls pause by setting playing=false or switching to 'latest'; clear the timer too, or it keeps stepping.
    if ((s.mapMode === 'latest' || !s.playing) && timer !== undefined) stop();
    render(s);
  });
  render(state);
}

import * as maplibregl from 'maplibre-gl';
import workerUrl from 'maplibre-gl/dist/maplibre-gl-worker.mjs?worker&url';
import { AOI, MAP_CENTER, MAP_ZOOM } from '../config.ts';
import { getChoropleth, getCommunes, getCompositeChoropleth, getScenes, getSummaryLayers } from '../data/dataSource.ts';

export interface MapView {

export function mountMap(container: HTMLElement): MapView {
  // maplibre-gl v6 locates its worker next to the library file, which breaks once bundled; point it at the bundled worker.
  maplibregl.setWorkerUrl(workerUrl);
  const map = new maplibregl.Map({
    container,
    style: baseStyle(),
    center: MAP_CENTER,
    zoom: MAP_ZOOM,
    minZoom: 3,
    maxBounds: [
      [90, -2],
      [125, 30],
    ],
    attributionControl: { compact: true },
    dragRotate: false,
    pitchWithRotate: false,
  });
  map.touchZoomRotate.disableRotation();
  map.addControl(new maplibregl.NavigationControl({ showCompass: false }), 'top-right');
  map.addControl(new maplibregl.ScaleControl({ unit: 'metric' }), 'bottom-right');
  map.addControl({
    onAdd() {
      const box = document.createElement('div');
      box.className = 'maplibregl-ctrl maplibregl-ctrl-group';
      const btn = document.createElement('button');
      btn.type = 'button';
      btn.className = 'ctrl-fit';
      btn.title = T.map.fitTip;
      btn.setAttribute('aria-label', T.map.fitTip);
      btn.textContent = '⌖';
      btn.addEventListener('click', () => map.fitBounds([[AOI[0], AOI[1]], [AOI[2], AOI[3]]], { padding: 30 }));
      box.append(btn);
      return box;
    },
    onRemove() {},
  }, 'top-right');

  new ResizeObserver(() => map.resize()).observe(container);
  if (import.meta.env.DEV) (window as unknown as { __map: maplibregl.Map }).__map = map;

  let scenes: Scene[] = [];
  let summaries: SummaryLayer[] = [];
  let communes: CommunesLayer | null = null;
  let collection: CommuneCollection | null = null;
  let images: SceneImages | null = null;
  let ready = false;
  let requestToken = 0;

  const sceneOf = (s: State) => scenes.find((x) => x.date === s.selectedDate && x.track === s.selectedTrack);

  function vis(id: string, on: boolean) {
    if (map.getLayer(id)) map.setLayoutProperty(id, 'visibility', on ? 'visible' : 'none');
  }

  function applyLook(s: State) {
    const summaryOn = s.mapMode === 'scene' && s.activeSummary !== null;
    const singleImage = s.mapMode === 'scene';
    const simple = s.viewMode === 'simple' && !!sceneOf(s)?.composite;
    vis('basemap', s.layers.basemap);
    images?.radar.setVisible(singleImage && s.layers.radar && !summaryOn && !simple);
    images?.water.setVisible(singleImage && s.layers.water && !summaryOn);
    vis('summary', summaryOn);
    vis('provinces', s.layers.provinces);
    vis('province-labels', s.layers.provinces);
    communes?.setVisible(s.layers.communes);
    images?.radar.setOpacity(s.opacity.radar);
    images?.water.setOpacity(s.opacity.water);
    // The simple view is one colour: commune fills stay clickable but transparent (outlines remain).
    communes?.setOpacity(simple && singleImage ? 0 : s.opacity.communes);
  }

  async function applyScene(s: State) {
    const token = ++requestToken;
    communes?.setMode(s.mapMode);
    if (s.mapMode === 'latest' && collection) {
      communes?.setChoropleth(latestValues(collection));
      return;
    }
    const scene = sceneOf(s);
    if (!scene) return;
    // Keep the previous values on screen until the next ones arrive (no grey flash between frames).
    set({ radarMissing: false });
    const urls = sceneImageUrls(scene, s.viewMode);
    void images?.radar.show(urls.radar, s.cut);
    void images?.water.show(urls.water, s.cut);
    try {
      const mosaic = scene.composite ? await getCompositeChoropleth(scene.date, scene.track) : null;
      if (token !== requestToken) return;
      if (mosaic) {
        communes?.setChoropleth(Object.fromEntries(Object.entries(mosaic).map(([id, v]) => [id, v.pct])), mosaic);
        return;
      }
      const values = await getChoropleth(scene.date, scene.track);
      if (token === requestToken) communes?.setChoropleth(values);
    } catch (err) {
      console.error(err);
    }
  }

  /** [Claude] Warm the HTTP cache so Play does not stall on the network (the simple view needs no radar). */
  function warmAll(mode: State['viewMode']) {
    for (const sc of scenes) {
      const u = sceneImageUrls(sc, mode);
      warmImage(u.water);
      if (mode === 'pro') warmImage(u.radar);
    }
  }

  function applySummary(s: State) {
    const layer = summaries.find((l) => l.id === s.activeSummary) ?? null;
    setSummaryLayer(map, layer);
  }

  function flyToCommune(id: string) {
    const b = communes?.bounds(id);
    if (!b) return;
    map.resize();
    const mobile = window.innerWidth < 768;
    map.fitBounds(b, {
      padding: { top: mobile ? 16 : 60, bottom: mobile ? Math.round(container.clientHeight * 0.62) + 8 : 40, left: mobile ? 16 : 40, right: mobile ? 16 : 480 },
      maxZoom: 11,
      duration: 700,
    });
  }

  map.on('error', (e) => {
    const err = e as unknown as { sourceId?: string };
    if (err.sourceId === 'radar-image') set({ radarMissing: true });
  });

  map.on('load', async () => {
    try {
      [scenes, summaries] = await Promise.all([getScenes(), getSummaryLayers()]);
      const first = sceneOf(state) ?? scenes[0];
      images = addImageLayers(map, first, state.viewMode, () => set({ radarMissing: true }));
      warmAll(state.viewMode);
      const data = await getCommunes();
      collection = data;
      communes = addCommunes(map, data, { onSelect: (osmId) => set({ selectedCommune: osmId }) });
      await addGeoLayers(map);
      ready = true;
      applyLook(state);
      if (state.activeSummary) applySummary(state);
      if (state.selectedCommune) communes.setSelected(state.selectedCommune);
      await applyScene(state);
      if (state.tab === 'map' && state.selectedCommune && state.flyNonce > 0) flyToCommune(state.selectedCommune);
    } catch (err) {
      console.error(err);
    }
  });

  subscribe((s, prev) => {
    if (s.tab === 'map' && prev.tab !== 'map') map.resize();
    if (!ready) return;
    if (s.selectedDate !== prev.selectedDate || s.selectedTrack !== prev.selectedTrack || s.mapMode !== prev.mapMode || s.viewMode !== prev.viewMode) {
      if (s.viewMode !== prev.viewMode) warmAll(s.viewMode);
      void applyScene(s);
    }
    if (s.layers !== prev.layers || s.opacity !== prev.opacity || s.activeSummary !== prev.activeSummary || s.mapMode !== prev.mapMode || s.viewMode !== prev.viewMode) {
      if (s.activeSummary !== prev.activeSummary) applySummary(s);
      applyLook(s);
    }
    if (s.selectedCommune !== prev.selectedCommune) communes?.setSelected(s.selectedCommune);
    if (s.flyNonce !== prev.flyNonce && s.selectedCommune) {
      flyToCommune(s.selectedCommune);
    }
  });

  return { map, resize: () => map.resize() };
}

/** [Codex] Explain the time and coverage of the visible map before users interpret it. */
export async function mountMapContext(host: HTMLElement, scenes: Scene[]): Promise<void> {
  const collection = await getCommunes();
  const assessed = collection.features.filter(f => hasAssessment(f.properties));
  const dates = assessed.map(f => f.properties.last_date!).sort();
  const title = h('h1');
  const note = h('p', { role: 'status', 'aria-live': 'polite' });
  const latest = h('button', { type: 'button', text: T.map.latestMode });
  const scene = h('button', { type: 'button', text: T.map.sceneMode });
  latest.addEventListener('click', () => set({ mapMode: 'latest', activeSummary: null, playing: false }));
  scene.addEventListener('click', () => set({ mapMode: 'scene', activeSummary: null, playing: false }));
  host.className = 'map-context';
  host.append(h('div', { class: 'map-context-heading' }, title,
    h('div', { class: 'map-mode', role: 'group', 'aria-label': T.map.modeLabel }, latest, scene)), note);
  let token = 0;
  async function render() {
    const my = ++token;
    const isLatest = state.mapMode === 'latest';
    latest.setAttribute('aria-pressed', String(isLatest));
    scene.setAttribute('aria-pressed', String(!isLatest));
    if (isLatest) {
      title.textContent = T.map.latestTitle;
      note.textContent = T.map.latestCoverage(assessed.length, collection.features.length, fmtDate(dates[0] ?? null), fmtDate(dates.at(-1) ?? null));
      return;
    }
    const current = scenes.find(s => s.date === state.selectedDate && s.track === state.selectedTrack);
    if (!current) return;
    title.textContent = T.map.sceneTitle(fmtDate(current.date), current.track);
    note.textContent = T.map.coverageLoading;
    try {
      if (current.composite) {
        const mosaic = await getCompositeChoropleth(current.date, current.track);
        if (mosaic) {
          if (my === token) note.textContent = T.map.blendCoverage(Object.keys(mosaic).length, collection.features.length);
          return;
        }
      }
      const values = await getChoropleth(current.date, current.track);
      if (my === token) note.textContent = T.map.sceneCoverage(Object.keys(values).length, collection.features.length);
    } catch {
      if (my === token) note.textContent = T.app.loadError;
    }
  }
  subscribe((s, prev) => {
    if (s.mapMode !== prev.mapMode || s.selectedDate !== prev.selectedDate || s.selectedTrack !== prev.selectedTrack) void render();
  });
  await render();
}

export type OverviewGroup = 'all' | 'high' | 'medium' | 'low' | 'insufficient';

type CoverageKind = 'sufficient' | 'partial' | 'none';

const INITIAL_VISIBLE_COUNT = 12;

function getCoverageKind(c: CommuneProps): CoverageKind {
  if (c.last_pct === null || c.last_date === null) return 'none';
  if (c.coverage_sufficient === true) return 'sufficient';
  return 'partial';
}

function isEvaluated(c: CommuneProps): boolean {
  return getCoverageKind(c) === 'sufficient';
}

function classifyCommune(c: CommuneProps): Exclude<OverviewGroup, 'all'> {
  if (!isEvaluated(c)) return 'insufficient';
  const pct = c.last_pct as number;
  if (pct >= 30) return 'high';
  if (pct >= 10) return 'medium';
  return 'low';
}

function comparePriority(a: CommuneProps, b: CommuneProps): number {
  const aEval = isEvaluated(a);
  const bEval = isEvaluated(b);
  if (aEval !== bEval) return aEval ? -1 : 1;

  const aKind = getCoverageKind(a);
  const bKind = getCoverageKind(b);
  if (aKind !== bKind) {
    if (aKind === 'partial') return -1;
    if (bKind === 'partial') return 1;
  }

  if (a.last_pct !== null && b.last_pct !== null && a.last_pct !== b.last_pct) {
    return b.last_pct - a.last_pct;
  }
  if (a.last_pct !== null && b.last_pct === null) return -1;
  if (a.last_pct === null && b.last_pct !== null) return 1;

  return a.name.localeCompare(b.name, 'vi');
}

function computeDateRange(all: CommuneProps[]): { minDate: string | null; maxDate: string | null } {
  const dates = all
    .map((c) => c.last_date)
    .filter((d): d is string => typeof d === 'string' && d.length > 0)
    .sort((a, b) => a.localeCompare(b));

  if (dates.length === 0) {
    return { minDate: null, maxDate: null };
  }
  return {
    minDate: dates[0],
    maxDate: dates[dates.length - 1],
  };
}

export function mountOverview(host: HTMLElement): void {
  host.classList.add('page', 'ov-page');
  const inner = h('div', { class: 'page-inner ov-inner' });
  host.replaceChildren(inner);
  inner.append(h('div', { class: 'loading ov-loading', text: T.overview.loading }));

  getCommunes()
    .then((collection) => {
      const propsList = collection.features.map((f) => f.properties);
      buildOverview(inner, propsList);
    })
    .catch((err: unknown) => {
      console.error(err);
      inner.replaceChildren(
        h('div', {
          class: 'error-box ov-error',
          text: err instanceof Error ? err.message : T.overview.loadError,
        }),
      );
    });
}

      { class: 'ov-section-head' },
      h('h2', { id: 'ov-list-heading', class: 'ov-section-title', text: T.overview.list.title }),
      h('p', { class: 'ov-section-sub', text: T.overview.list.subtitle }),
    ),
    h(
      'div',
      { class: 'ov-toolbar' },
      searchInput,
      provinceSelect,
      groupSelect,
      resetFilterBtn,
      h('div', { class: 'ov-toolbar-spacer' }),
      resultSummary,
    ),
    h('div', { class: 'ov-table-wrap' }, table),
    listFooter,
  );

  // 4. Data Caveats & 3-Step Guide Section
  const exAnLong =
    all.find((c) => c.osm_id === '15581562') ??
    all.find((c) => isEvaluated(c) && c.reference_track === '083A');
  const exSonKien =
    all.find((c) => c.osm_id === '15416586') ??
    all.find((c) => isEvaluated(c) && c.reference_track === '148D');

  const trackExampleText =
    exAnLong && exSonKien
      ? T.overview.caveats.trackExampleDynamic(
          exAnLong.name,
          fmtPct(exAnLong.last_pct),
          fmtDate(exAnLong.last_date),
          exAnLong.reference_track ?? '083A',
          exSonKien.name,
          fmtPct(exSonKien.last_pct),
          fmtDate(exSonKien.last_date),
          exSonKien.reference_track ?? '148D',
        )
      : T.overview.caveats.trackExampleFallback;

  const filterInsufficientQuickBtn = h('button', {
    type: 'button',
    class: 'btn ov-btn-inline',
    text: T.overview.caveats.viewInsufficientBtn(insufficientList.length),
  });
  filterInsufficientQuickBtn.addEventListener('click', () => {
    activeGroup = 'insufficient';
    groupSelect.value = 'insufficient';
    expanded = true;
    renderList();
    listSection.scrollIntoView({ behavior: 'smooth', block: 'start' });
  });

  const guideAndCaveatsSection = h(
    'section',
    { class: 'ov-bottom-grid', 'aria-label': T.overview.guideAndCaveatsAria },
    h(
      'div',
      { class: 'ov-panel ov-caveats-panel' },
      h('h2', { class: 'ov-panel-title', text: T.overview.caveats.title }),
      h(
        'div',
        { class: 'ov-caveat-block' },
        h('h3', {
          class: 'ov-caveat-heading',
          text: T.overview.caveats.missingNotDryTitle(insufficientList.length),
        }),
        h('p', {
          class: 'ov-caveat-text',
          text: T.overview.caveats.missingNotDryBody(insufficientList.length, noObsCount, partialObsCount),
        }),
        filterInsufficientQuickBtn,
      ),
      h(
        'div',
        { class: 'ov-caveat-block' },
        h('h3', { class: 'ov-caveat-heading', text: T.overview.caveats.multiTrackTitle }),
        h('p', { class: 'ov-caveat-text', text: T.overview.caveats.multiTrackBody }),
        h('p', { class: 'ov-caveat-example', text: trackExampleText }),
      ),
    ),
    h(
      'div',
      { class: 'ov-panel ov-guide-panel' },
      h('h2', { class: 'ov-panel-title', text: T.overview.guide.title }),
      h('p', { class: 'ov-panel-lead', text: T.overview.guide.lead }),
      h(
        'ol',
        { class: 'ov-steps' },
        h(
          'li',
          { class: 'ov-step' },
          h('strong', { class: 'ov-step-title', text: T.overview.guide.step1Title }),
          h('p', { class: 'ov-step-body', text: T.overview.guide.step1Body }),
        ),
        h(
          'li',
          { class: 'ov-step' },
          h('strong', { class: 'ov-step-title', text: T.overview.guide.step2Title }),
          h('p', { class: 'ov-step-body', text: T.overview.guide.step2Body }),
        ),
        h(
          'li',
          { class: 'ov-step' },
          h('strong', { class: 'ov-step-title', text: T.overview.guide.step3Title }),
          h('p', { class: 'ov-step-body', text: T.overview.guide.step3Body }),
        ),
      ),
    ),
  );

  root.replaceChildren(heroSection, cardsSection, listSection, h('p', { class: 'note ov-footnote', text: T.overview.cautionBody }), h('details', { class: 'ov-explanation' }, h('summary', { text: T.overview.caveats.title }), guideAndCaveatsSection));

  function getFilteredRows(): CommuneProps[] {
    const q = fold(searchQuery.trim());
    return all
      .filter((c) => {
        if (selectedProvince && c.province !== selectedProvince) return false;
        if (activeGroup !== 'all' && classifyCommune(c) !== activeGroup) return false;
        if (q && !fold(c.name).includes(q) && !fold(c.province).includes(q)) return false;
        return true;
      })
      .sort(comparePriority);
  }

  function renderWaterCell(c: CommuneProps): HTMLElement {
    const kind = getCoverageKind(c);
    if (kind === 'none' || c.last_pct === null) {
      return h(
        'td',
        { class: 'ov-num' },
        h('span', { class: 'ov-pct-empty', text: '—' }),
        h('span', { class: 'ov-pct-sub', text: T.overview.list.noObsShort }),
      );
    }

    const pctText = fmtPct(c.last_pct);
    const group = classifyCommune(c);
    const widthPct = Math.max(2, Math.min(100, Math.round(c.last_pct)));

    return h(
      'td',
      { class: 'ov-num' },
      h(
        'div',
        { class: 'ov-pct-wrap' },
        h('strong', { class: `ov-pct-val ov-pct-${group}`, text: pctText }),
        kind === 'partial'
          ? h('span', { class: 'ov-pct-partial-tag', text: T.overview.list.partialValueNote })
          : h('span', { class: 'ov-pct-group-tag', text: T.overview.list.groupBadge[group] }),
      ),
      h(
        'div',
        { class: 'ov-bar-track', 'aria-hidden': 'true' },
        h('div', { class: `ov-bar-fill ov-bar-${group}`, style: `width:${widthPct}%` }),
      ),
    );
  }

  function renderDateCell(c: CommuneProps): HTMLElement {
    if (!c.last_date) {
      return h('td', {}, h('span', { class: 'ov-date-empty', text: T.overview.list.noDate }));
    }
    return h(
      'td',
      {},
      h('span', { class: 'ov-date-val', text: fmtDate(c.last_date) }),
    );
  }

  function renderCoverageCell(c: CommuneProps): HTMLElement {
    const kind = getCoverageKind(c);
    if (kind === 'sufficient') {
      return h(
        'td',
        {},
        h('span', { class: 'ov-cov-chip ov-cov-ok', text: T.overview.list.coverageSufficient }),
        c.valid_km2 !== null
          ? h('span', { class: 'ov-cov-sub', text: T.overview.list.validArea(fmtNum(c.valid_km2)) })
          : null,
      );
    }
    if (kind === 'partial') {
      return h(
        'td',
        {},
        h('span', { class: 'ov-cov-chip ov-cov-partial', text: T.overview.list.coveragePartial }),
        c.valid_km2 !== null
          ? h('span', { class: 'ov-cov-sub', text: T.overview.list.validArea(fmtNum(c.valid_km2)) })
          : null,
      );
    }
    return h(
      'td',
      {},
      h('span', { class: 'ov-cov-chip ov-cov-none', text: T.overview.list.coverageNone }),
    );
  }

  function renderList(): void {
    for (const [grp, btn] of cardButtons.entries()) {
      const isPressed = activeGroup === grp;
      btn.setAttribute('aria-pressed', isPressed ? 'true' : 'false');
      btn.classList.toggle('is-active', isPressed);
    }

    const hasFilters = activeGroup !== 'all' || selectedProvince !== '' || searchQuery.trim() !== '';
    resetFilterBtn.disabled = !hasFilters;

    const filtered = getFilteredRows();
    const visible = expanded ? filtered : filtered.slice(0, INITIAL_VISIBLE_COUNT);

    resultSummary.textContent = T.overview.list.showingCount(visible.length, filtered.length, totalCount);

    if (filtered.length === 0) {
      tbody.replaceChildren(
        h(
          'tr',
          {},
          h('td', { colspan: '6', class: 'ov-empty', text: T.overview.list.emptyResult }),
        ),
      );
      listFooter.hidden = true;
      return;
    }

    tbody.replaceChildren(
      ...visible.map((c) => {
        const kind = getCoverageKind(c);
        const covLabel =
          kind === 'sufficient'
            ? T.overview.list.coverageSufficient
            : kind === 'partial'
              ? T.overview.list.coveragePartial
              : T.overview.list.coverageNone;

        const rowAria = T.overview.list.rowAria(
          c.name,
          c.province,
          fmtPct(c.last_pct),
          c.last_date ? fmtDate(c.last_date) : T.overview.list.noDate,
          covLabel,
        );

        const openBtn = h('button', {
          type: 'button',
          class: 'ov-row-btn',
          text: T.overview.list.openMapRow,
          'aria-label': rowAria,
        });
        openBtn.addEventListener('click', (ev) => {
          ev.stopPropagation();
          openCommuneOnMap(c);
        });

        const tr = h(
          'tr',
          {
            class: `ov-row ov-row-${classifyCommune(c)}`,
          },
          h('td', { class: 'ov-cell-name' }, h('strong', { text: c.name })),
          h('td', { class: 'ov-cell-prov', text: c.province }),
          renderWaterCell(c),
          renderDateCell(c),
          renderCoverageCell(c),
          h('td', { class: 'ov-col-action' }, openBtn),
        );

        tr.addEventListener('click', () => openCommuneOnMap(c));


        return tr;
      }),
    );

    if (filtered.length > INITIAL_VISIBLE_COUNT) {
      listFooter.hidden = false;
      const remaining = filtered.length - INITIAL_VISIBLE_COUNT;
      showMoreBtn.textContent = expanded
        ? T.overview.list.showLess
        : T.overview.list.showMore(remaining);
    } else {
      listFooter.hidden = true;
    }
  }

  searchInput.addEventListener('input', () => {
    searchQuery = searchInput.value;
    expanded = false;
    renderList();
  });

  provinceSelect.addEventListener('change', () => {
    selectedProvince = provinceSelect.value;
    expanded = false;
    renderList();
  });

  groupSelect.addEventListener('change', () => {
    activeGroup = groupSelect.value as OverviewGroup;
    expanded = false;
    renderList();
  });

  resetFilterBtn.addEventListener('click', () => {
    activeGroup = 'all';
    selectedProvince = '';
    searchQuery = '';
    searchInput.value = '';
    provinceSelect.value = '';
    groupSelect.value = 'all';
    expanded = false;
    renderList();
  });

  showMoreBtn.addEventListener('click', () => {
    expanded = !expanded;
    renderList();
  });

  renderList();
}
